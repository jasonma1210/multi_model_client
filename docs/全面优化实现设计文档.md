```dart
class ModelPathCache {
  static final ModelPathCache _instance = ModelPathCache._();
  final Map<String, String> _pathIndex = {};  // fileName -> fullPath
  bool _initialized = false;

  Future<String?> findModel(String fileName) async {
    if (!_initialized) await _buildIndex();
    return _pathIndex[fileName];
  }

  Future<void> _buildIndex() async {
    final dirs = await _getAllModelDirectories();
    for (final dir in dirs) {
      await for (final entity in dir.list(recursive: true)) {
        if (entity is File && entity.path.endsWith('.gguf')) {
          _pathIndex[p.basename(entity.path)] = entity.path;
        }
      }
    }
    _initialized = true;
  }

  void invalidate() => _initialized = false;
}
```

### 1.2 Token检测速度优化

**目标**: 解决3-4轮对话后的降速问题

#### 1.2.1 当前问题

- `_cleanThinkTags` 每个token执行5次正则替换
- Token估算算法粗糙（中文*1.5 + 其他/4）
- 压缩触发时机不精确

#### 1.2.2 优化方案

**方案A: 正则预编译 + 流式缓冲**

```dart
class TokenProcessor {
  static final _thinkTagPattern = RegExp(r'<think>[\s\S]*?</think>');
  static final _partialThinkPattern = RegExp(r'<think>[\s\S]*$');
  
  String _buffer = '';
  
  String processChunk(String chunk) {
    _buffer += chunk;
    
    // 完整标签移除
    if (_thinkTagPattern.hasMatch(_buffer)) {
      _buffer = _buffer.replaceAll(_thinkTagPattern, '');
    }
    
    // 部分标签检测（流式中可能不完整）
    if (_partialThinkPattern.hasMatch(_buffer)) {
      return '';  // 暂时隐藏，等待完整标签
    }
    
    final result = _buffer;
    _buffer = '';
    return result;
  }
}
```

**方案B: 精确Token计数**

```dart
class AccurateTokenCounter {
  // 针对不同模型的token比率
  static const _chineseRatio = 1.8;  // 中文字符平均token数
  static const _englishRatio = 0.25; // 英文单词平均token数
  static const _codeRatio = 0.4;     // 代码字符平均token数
  static const _specialRatio = 0.5;  // 特殊符号平均token数
  
  static int estimate(String text) {
    int count = 0;
    for (int i = 0; i < text.length; i++) {
      final char = text[i];
      if (_isChinese(char)) {
        count += _chineseRatio.ceil();
      } else if (_isAlphaNumeric(char)) {
        count += 1;  // 英文字符按字符计
      } else {
        count += _specialRatio.ceil();
      }
    }
    return count;
  }
}
```

### 1.3 TTS同步优化

**目标**: AI回复完成后TTS立即开始播放

#### 1.3.1 当前问题

- 长文本先全部合成再播放（延迟30+秒）
- 句子分割后逐句合成，块间有停顿
- Android TTS初始化轮询开销大

#### 1.3.2 优化方案

**方案A: 流式TTS管道**

```dart
class StreamingTTSManager {
  final _synthesisQueue = Queue<AudioChunk>();
  final _playbackController = StreamController<AudioChunk>();
  
  void startStreaming() {
    // 播放器持续从队列消费
    _playbackController.stream.listen((chunk) {
      _playAudioChunk(chunk);
    });
  }
  
  void onTokenGenerated(String token) {
    _sentenceBuffer += token;
    
    // 检测句子边界
    if (_isSentenceEnd(_sentenceBuffer)) {
      final sentence = _sentenceBuffer;
      _sentenceBuffer = '';
      
      // 异步合成，不阻塞token生成
      _synthesizeAndQueue(sentence);
    }
  }
  
  Future<void> _synthesizeAndQueue(String text) async {
    final audio = await _ttsService.synthesize(text);
    _synthesisQueue.add(AudioChunk(audio));
  }
}
```

**方案B: 双缓冲播放**

```dart
class DoubleBufferPlayer {
  AudioChunk? _currentBuffer;
  AudioChunk? _nextBuffer;
  
  Future<void> playChunk(AudioChunk chunk) async {
    _nextBuffer = chunk;
    
    if (_currentBuffer == null) {
      _currentBuffer = _nextBuffer;
      _nextBuffer = null;
      await _playBuffer(_currentBuffer!);
    }
    // 当前缓冲播放完后自动切换到nextBuffer
  }
  
  void _onCurrentComplete() {
    if (_nextBuffer != null) {
      _currentBuffer = _nextBuffer;
      _nextBuffer = null;
      _playBuffer(_currentBuffer!);
    }
  }
}
```

## 2. 上下文记忆功能重构设计

### 2.1 当前架构问题

| 问题 | 描述 |
|------|------|
| Token估算粗糙 | 中文*1.5 + 其他/4，误差超50% |
| 压缩策略单一 | 仅滑动窗口+摘要，无优先级区分 |
| 记忆检索阻塞 | 同步等待记忆宫殿检索 |
| 数据库逐条写入 | 压缩后逐条INSERT，效率低 |

### 2.2 重构方案

#### 2.2.1 分层上下文管理

```
┌─────────────────────────────────────────────────────────┐
│                    Context Manager                        │
├─────────────────────────────────────────────────────────┤
│  Layer 1: System Prompt (固定，不压缩)                    │
│  Layer 2: Long-term Memory (记忆宫殿，重要性加权)         │
│  Layer 3: Session Summary (会话摘要，定期更新)            │
│  Layer 4: Recent Messages (滑动窗口，最近N条)             │
│  Layer 5: Current Context (当前对话，实时)                │
└─────────────────────────────────────────────────────────┘
```

#### 2.2.2 智能压缩算法

```dart
class SmartContextCompressor {
  // 压缩策略优先级
  static const _priorityWeights = {
    'system': 1.0,      // 系统消息永不压缩
    'tool_result': 0.9,  // 工具结果高优先
    'user_query': 0.8,   // 用户查询高优先
    'assistant': 0.6,    // AI回复中优先
    'memory': 0.5,       // 记忆中优先
    'summary': 0.3,      // 摘要低优先
  };
  
  Future<List<Message>> compress({
    required List<Message> messages,
    required int targetTokens,
    required int currentTokens,
  }) async {
    // 1. 计算每条消息的重要性分数
    final scored = _scoreMessages(messages);
    
    // 2. 按重要性排序，保留高优先级
    scored.sort((a, b) => b.score.compareTo(a.score));
    
    // 3. 低优先级消息合并为摘要
    final toCompress = scored.where((s) => s.score < 0.5).toList();
    final summary = await _generateSummary(toCompress);
    
    // 4. 组装最终上下文
    return _assembleContext(scored, summary, targetTokens);
  }
}
```

```dart
class AsyncMemoryRetriever {
  final _cache = <String, CachedMemory>{};
  final _cacheTimeout = Duration(minutes: 5);
  
  Future<MemoryContext> retrieve(String query, String sessionId) async {
    // 1. 检查缓存
    final cached = _cache[sessionId];
    if (cached != null && !cached.isExpired) {
      return cached.context;
    }
    
    // 2. 并行检索
    final results = await Future.wait([
      _searchByKeywords(query),
      _searchBySemantics(query),
      _searchByRecency(sessionId),
    ]);
    
    // 3. 合并排序
    final merged = _mergeResults(results);
    
    // 4. 缓存结果
    _cache[sessionId] = CachedMemory(
      context: merged,
      timestamp: DateTime.now(),
    );
    
    return merged;
  }
}
```

```dart
class DocumentGenerationService {

  Future<File> generatePDF({
    required String title,
    required List<DocumentSection> sections,

  }) async {
    final pdf = pw.Document();
    
    // 添加封面

    for (final section in sections) {
      pdf.addPage(_buildSectionPage(section));
    }
    
    // 保存文件
    final file = await _saveToFile(pdf, '$title.pdf');
    return file;

  /// 生成Excel文档
  Future<File> generateExcel({
    required String title,
    required List<SheetData> sheets,
  }) async {
    final excel = Excel.createExcel();
    
    for (final sheet in sheets) {
      final table = excel[sheet.name];
      
      // 添加表头
      table.appendRow(sheet.headers);
      
      // 添加数据行
      for (final row in sheet.rows) {
        table.appendRow(row);
      }
    }
    
    return _saveToFile(excel, '$title.xlsx');
  }
  
  /// 生成PPT文档

    required String title,
    required List<SlideData> slides,
  }) async {
    // PPTX是ZIP格式，包含XML文件

    _addRelationships(archive);
    _addPresentation(archive, slides);
    
    return _saveArchiveToFile(archive, '$title.pptx');
```

### 5.5 异步转录机制

```dart
class AsyncTranscriptionService {
  final _taskQueue = Queue<TranscriptionTask>();
  final _resultController = StreamController<TranscriptionResult>();
  
  Stream<TranscriptionResult> get results => _resultController.stream;
  
  Future<void> submitTask(AudioSegment segment) async {
    final task = TranscriptionTask(
      id: const Uuid().v4(),
      segment: segment,
      status: TaskStatus.pending,
    );
    
    _taskQueue.add(task);
    _processQueue();
  }
  
  Future<void> _processQueue() async {
    if (_taskQueue.isEmpty) return;
    
    final task = _taskQueue.removeFirst();
    task.status = TaskStatus.processing;
    
    try {
      final transcription = await _asrService.recognize(task.segment.filePath);
      
      task.status = TaskStatus.completed;
      _resultController.add(TranscriptionResult(
        taskId: task.id,
        transcription: transcription,
      ));
      
      // 显示通知
      _showCompletionNotification(task);
      
    } catch (e) {
      task.status = TaskStatus.failed;
      _resultController.addError(e);
    }
    
    // 继续处理队列
    _processQueue();
  }
}
```

```dart
class MimoVoiceCloneService {
  static const _baseUrl = 'https://api.xiaomimimo.com/v1';
  static const _model = 'mimo-v2.5-tts-voiceclone';
  
  final Dio _dio;
  
  MimoVoiceCloneService() : _dio = Dio(BaseOptions(
    baseUrl: _baseUrl,
    connectTimeout: Duration(seconds: 30),
    receiveTimeout: Duration(minutes: 5),
  ));
  
  /// 提交语音克隆任务
  Future<CloneTaskResult> submitCloneTask({
    required File audioFile,
    required String voiceName,
    String? testText,
  }) async {
    // 1. 读取并编码音频
    final audioBytes = await audioFile.readAsBytes();
    final base64Audio = base64Encode(audioBytes);
    final mimeType = _getMimeType(audioFile.path);
    
    // 2. 构建请求
    final response = await _dio.post('/chat/completions', data: {
      'model': _model,
      'messages': [
        {
          'role': 'user',
          'content': testText ?? '这是一段测试文本，用于验证语音克隆效果。',
        }
      ],
      'audio': {
        'format': _getAudioFormat(audioFile.path),
        'voice': 'data:$mimeType;base64,$base64Audio',
      },
    });
    
    // 3. 解析响应
    return CloneTaskResult.fromJson(response.data);
  }
  
  /// 测试克隆语音
  Future<File> testClonedVoice({
    required String clonedVoiceId,
    required String text,
  }) async {
    final response = await _dio.post('/chat/completions', data: {
      'model': _model,
      'messages': [
        {'role': 'user', 'content': text}
      ],
      'voice_id': clonedVoiceId,
    });
    
    // 保存合成音频
    final audioData = base64Decode(response.data['audio']['data']);
```

```dart
    _micSubscription = stream.listen((chunk) {
```

#### 8.2.2 实时VAD检测器

```dart
class RealtimeVADDetector {
  final double threshold;
  final Duration silenceTimeout;
  
  bool _isSpeaking = false;
  DateTime? _lastSpeechTime;
  final _speechController = StreamController<VADEvent>();
  
  Stream<VADEvent> get events => _speechController.stream;
  
  void processAudioChunk(Uint8List chunk) {
    final energy = _calculateEnergy(chunk);
    final now = DateTime.now();
    
    if (energy > threshold) {
      if (!_isSpeaking) {
        _isSpeaking = true;
        _speechController.add(VADEvent.speechStart);
      }
      _lastSpeechTime = now;
    } else if (_isSpeaking && _lastSpeechTime != null) {
      if (now.difference(_lastSpeechTime!) > silenceTimeout) {
        _isSpeaking = false;
        _speechController.add(VADEvent.speechEnd);
      }
    }
  }
  
  double _calculateEnergy(Uint8List chunk) {
    double sum = 0;
    for (int i = 0; i < chunk.length; i += 2) {
      final sample = chunk[i] | (chunk[i + 1] << 8);
      sum += sample * sample;
    }
    return sqrt(sum / (chunk.length / 2));
  }
}
```

### 8.3 主控制器

```dart
class RealtimeVoiceDialogueController extends StateNotifier<VoiceDialogueState> {
  final AudioStreamManager _audioManager;
  final ASRService _asrService;
  final InferenceEngineManager _llmEngine;
  final TTSService _ttsService;
  final RealtimeVADDetector _vadDetector;
  final VoiceInterruptionManager _interruptionManager;
  
  StreamSubscription<Uint8List>? _audioSubscription;
  StreamSubscription<VADEvent>? _vadSubscription;
  StreamSubscription<void>? _interruptionSubscription;
  
  final _textBuffer = StringBuffer();
  bool _isProcessing = false;
  
  Future<void> startDialogue(String sessionId) async {
    state = VoiceDialogueState.listening;
    
    // 启动音频捕获
    await _audioManager.startCapture();
    
    // 监听音频流
    _audioSubscription = _audioManager.audioStream.listen((chunk) {
      _vadDetector.processAudioChunk(chunk);
    });
    
    // 监听VAD事件
    _vadSubscription = _vadDetector.events.listen((event) {
      switch (event) {
        case VADEvent.speechStart:
          _onSpeechStart();
          break;
        case VADEvent.speechEnd:
          _onSpeechEnd();
          break;
      }
    });
    
    // 监听打断事件
    _interruptionSubscription = _interruptionManager.interruptions.listen((_) {
      _onInterruption();
    });
  }
  
  Future<void> _onSpeechEnd() async {
    if (_isProcessing) return;
    _isProcessing = true;
    
    state = VoiceDialogueState.processing;
    
    // ASR识别
    final text = await _asrService.recognize(_audioBuffer);
    
    // LLM推理（流式）
    state = VoiceDialogueState.thinking;
    final responseStream = _llmEngine.streamResponse(text);
    
    // 流式TTS
    state = VoiceDialogueState.speaking;
    _interruptionManager.onModelStartSpeaking();
    
    await for (final chunk in responseStream) {
      _ttsService.speakChunk(chunk);
    }
    
    _interruptionManager.onModelStopSpeaking();
    state = VoiceDialogueState.listening;
    _isProcessing = false;
  }
  
  void _onInterruption() {
    // 停止TTS播放
    _ttsService.stop();
    
    // 停止当前推理
    _llmEngine.cancelCurrent();
    
    _interruptionManager.onModelStopSpeaking();
    state = VoiceDialogueState.listening;
    _isProcessing = false;
  }
  
  Future<void> stopDialogue() async {
    await _audioSubscription?.cancel();
    await _vadSubscription?.cancel();
    await _interruptionSubscription?.cancel();
    await _audioManager.stopCapture();
    await _ttsService.stop();
    
    state = VoiceDialogueState.idle;
  }
}
```
